cmake_minimum_required(VERSION 3.22)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(MujocoDirectRotorSystem LANGUAGES CXX)
  find_package(gz-sim8 REQUIRED)
  find_package(gz-transport13 REQUIRED)
  find_package(gz-plugin2 REQUIRED COMPONENTS register)
  set(GZ_SIM_TARGET gz-sim8::gz-sim8)
  set(GZ_TRANSPORT_TARGET gz-transport13::gz-transport13)
  set(GZ_PLUGIN_TARGET gz-plugin2::gz-plugin2)
endif()
add_library(MujocoDirectRotorSystem SHARED MujocoDirectRotorSystem.cpp)
target_compile_features(MujocoDirectRotorSystem PRIVATE cxx_std_17)
target_link_libraries(MujocoDirectRotorSystem PRIVATE ${GZ_SIM_TARGET} ${GZ_TRANSPORT_TARGET} ${GZ_PLUGIN_TARGET})
install(TARGETS MujocoDirectRotorSystem LIBRARY DESTINATION lib/px4_gz_plugins)
