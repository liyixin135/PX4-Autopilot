cmake_minimum_required(VERSION 3.22)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(SelectiveDeltaCollisionSystem LANGUAGES CXX)
  find_package(gz-sim8 REQUIRED)
  find_package(gz-plugin2 REQUIRED COMPONENTS register)
  set(GZ_SIM_TARGET gz-sim8::gz-sim8)
  set(GZ_PLUGIN_TARGET gz-plugin2::gz-plugin2)
endif()
add_library(SelectiveDeltaCollisionSystem SHARED SelectiveDeltaCollisionSystem.cpp)
target_compile_features(SelectiveDeltaCollisionSystem PRIVATE cxx_std_17)
target_link_libraries(SelectiveDeltaCollisionSystem PRIVATE ${GZ_SIM_TARGET} ${GZ_PLUGIN_TARGET})
install(TARGETS SelectiveDeltaCollisionSystem LIBRARY DESTINATION lib/px4_gz_plugins)
